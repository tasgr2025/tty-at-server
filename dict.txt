# dict.txt 
# Регулярные выражения для команд и ответов.
# Ответ начинается после крайнего правого символа `=`.
# Левая часть регулярное выражение с ограниченым синтаксисом.
# В квадратных скобках указывают один из допустимых символов - ATE[1,0]
# значит множество ATE0, ATE1. Точка - любой один символ.
# Звёздочка - ни одного или несколько любых символов.
AT=OK
ATE[1,0]=OK
ATI=Manufacturer: Test, Model: Test, Revision: 1.0
AT+COPS?=+COPS: 0,0,"TestNet",0
AT+CPIN?=+CPIN: READY
AT+CPIN=*=OK
ATDF.=OK
AT+CGDCONT=*=OK
AT+CGACT=*=OK
ATZ=OK
AT*A=ERROR
