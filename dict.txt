# dict.txt 
# Регулярные выражения для команд и ответов.
# Ответ начинается после крайнего правого символа `=`.
# Левая часть регулярное выражение с ограниченым синтаксисом:
# квадратных скобках указывают один допустимых символов - ATE[1,0] значит множество ATE0, ATE1.
# Точка любой один символ.
AT=OK
ATE[1,0]=OK
ATI=Manufacturer: Test, Model: Test, Revision: 1.0
AT+COPS?=+COPS: 0,0,"TestNet",0
AT+CPIN?=+CPIN: READY
AT+CPIN=*=OK
AT*B=OK
ATDF.=OK
